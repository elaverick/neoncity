# --- Shader compiler: use the Vulkan SDK's if present, otherwise build glslang -
find_program(GLSLC glslc HINTS $ENV{VULKAN_SDK}/bin $ENV{VULKAN_SDK}/Bin)
if(GLSLC)
  message(STATUS "Using glslc: ${GLSLC}")
  set(SHADER_CMD ${GLSLC} --target-env=vulkan1.3 -O -mfmt=num)
  set(SHADER_DEP "")
else()
  find_program(GLSLANG_VALIDATOR glslangValidator HINTS $ENV{VULKAN_SDK}/bin $ENV{VULKAN_SDK}/Bin)
  if(GLSLANG_VALIDATOR)
    message(STATUS "Using glslangValidator: ${GLSLANG_VALIDATOR}")
    set(SHADER_CMD ${GLSLANG_VALIDATOR} -V --target-env vulkan1.3 -x)
    set(SHADER_DEP "")
  else()
    message(STATUS "No Vulkan SDK shader compiler found - building glslang from source")
    set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
    set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
    set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
    set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
    set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
    set(ENABLE_GLSLANG_BINARIES ON CACHE BOOL "" FORCE)
    FetchContent_Declare(glslang
      GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git
      GIT_TAG 15.1.0 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glslang)
    set(SHADER_CMD $<TARGET_FILE:glslang-standalone> -V --target-env vulkan1.3 -x)
    set(SHADER_DEP glslang-standalone)
  endif()
endif()

# --- Materials -------------------------------------------------------------------
# Each assets/materials/<name>.json defines an authored PBR material (src/render/material_def.h).
# Its id, PBR_<NAME>, is its place among the files in name order: written here into
# pbr_materials.h, which C and the shaders share (src/core/pbr.h), with the names for C in
# pbr_material_names.inc. Adding or removing a file reconfigures (CONFIGURE_DEPENDS); the
# files are only rewritten when the list changes.
file(GLOB MATERIAL_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/assets/materials/*.json)
list(SORT MATERIAL_FILES)
if(NOT MATERIAL_FILES)
  message(FATAL_ERROR "No materials in assets/materials/")
endif()
set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(PBR_IDS "")
set(PBR_NAMES "")
set(PBR_COUNT 0)
foreach(F ${MATERIAL_FILES})
  get_filename_component(N ${F} NAME)
  string(REGEX REPLACE "\\.json$" "" N ${N})
  if(NOT N MATCHES "^[a-z][a-z0-9_]*$")
    message(FATAL_ERROR "Material file ${F}: names are lower case letters, digits and underscores")
  endif()
  string(TOUPPER ${N} U)
  string(APPEND PBR_IDS "#define PBR_${U} ${PBR_COUNT}\n")
  string(APPEND PBR_NAMES "    \"${N}\",\n")
  math(EXPR PBR_COUNT "${PBR_COUNT} + 1")
endforeach()
file(CONFIGURE OUTPUT ${GEN_DIR}/pbr_materials.h CONTENT
"/* Generated by CMake from assets/materials/*.json (CMakeLists.txt, \"Materials\"): don't edit.
 * Each material's id is its file's place in name order. #defines only: shared by C and GLSL (src/core/pbr.h). */
#ifndef PBR_MATERIALS_H
#define PBR_MATERIALS_H
@PBR_IDS@#define PBR_MATERIAL_COUNT ${PBR_COUNT}
#endif
" @ONLY)
file(CONFIGURE OUTPUT ${GEN_DIR}/pbr_material_names.inc CONTENT
"/* Generated by CMake from assets/materials/*.json: the materials' names in id order (PBR_MATERIAL_NAMES). */
@PBR_NAMES@" @ONLY)

# Shaders are compiled to comma-separated SPIR-V word lists and #included into
# the executable, so the game is a single self-contained binary.
set(SHADER_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
set(SHADER_OUT_DIR ${CMAKE_CURRENT_BINARY_DIR}/spv)
file(MAKE_DIRECTORY ${SHADER_OUT_DIR} ${SHADER_OUT_DIR}/terrain ${SHADER_OUT_DIR}/proxy)
set(SHADERS city.vert city.frag fullscreen.vert sky.frag composite.frag bloom_down.comp bloom_up.comp
            cloud_noise.comp clouds.comp cloud_composite.frag ssr.comp ssr_apply.frag shelter.vert shelter.frag
            prepass.frag shadow_trace.comp shadow_denoise.comp terrain/terrain.vert terrain/terrain.frag
            terrain/planet.vert terrain/planet.frag
            proxy/proxy.vert proxy/proxy.frag flame.vert flame.frag smoke.vert smoke.frag)
# Variants of a shader built with a define: output:source:define. Ray-traced
# shadows need ray queries in the scene pass, and a depth prepass.
set(SHADER_VARIANTS city_rt.frag:city.frag:RT_SHADOWS city_pre.vert:city.vert:PREPASS)
# Shaders also include the #define-only headers they share with C.
set(SHADER_SHARED_HEADERS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/distances.h ${CMAKE_CURRENT_SOURCE_DIR}/src/core/clouds.h
                          ${CMAKE_CURRENT_SOURCE_DIR}/src/terrain/terrain_clip.h
                          ${CMAKE_CURRENT_SOURCE_DIR}/src/terrain/terrain_planet.h
                          ${CMAKE_CURRENT_SOURCE_DIR}/src/core/pbr.h ${GEN_DIR}/pbr_materials.h)
set(SHADER_SHARED_INC -I${CMAKE_CURRENT_SOURCE_DIR}/src/core -I${CMAKE_CURRENT_SOURCE_DIR}/src/terrain -I${GEN_DIR})
file(GLOB_RECURSE SHADER_INCLUDES ${SHADER_SRC_DIR}/*.glsl)
list(APPEND SHADER_INCLUDES ${SHADER_SHARED_HEADERS})
set(SPV_OUTPUTS "")
foreach(S ${SHADERS})
  set(OUT ${SHADER_OUT_DIR}/${S}.inc)
  add_custom_command(OUTPUT ${OUT}
    COMMAND ${SHADER_CMD} -I${SHADER_SRC_DIR} ${SHADER_SHARED_INC} -o ${OUT} ${SHADER_SRC_DIR}/${S}
    DEPENDS ${SHADER_SRC_DIR}/${S} ${SHADER_INCLUDES} ${SHADER_DEP}
    COMMENT "Compiling shader ${S}" VERBATIM)
  list(APPEND SPV_OUTPUTS ${OUT})
endforeach()
foreach(V ${SHADER_VARIANTS})
  string(REPLACE ":" ";" P ${V})
  list(GET P 0 NAME)
  list(GET P 1 SRC)
  list(GET P 2 DEF)
  set(OUT ${SHADER_OUT_DIR}/${NAME}.inc)
  add_custom_command(OUTPUT ${OUT}
    COMMAND ${SHADER_CMD} -I${SHADER_SRC_DIR} ${SHADER_SHARED_INC} -D${DEF}=1 -o ${OUT} ${SHADER_SRC_DIR}/${SRC}
    DEPENDS ${SHADER_SRC_DIR}/${SRC} ${SHADER_INCLUDES} ${SHADER_DEP}
    COMMENT "Compiling shader ${NAME}" VERBATIM)
  list(APPEND SPV_OUTPUTS ${OUT})
endforeach()
add_custom_target(shaders DEPENDS ${SPV_OUTPUTS})

# --- Tools: the Earth bake ------------------------------------------------------
# Bakes the Earth project's staged grid (tools/earth/earth.py stage) onto the six cube
# faces the engine reads (tools/earth_bake.c, data/earth/SOURCE.md). It links the
# cube-sphere alone, so its directions are the engine's bit for bit.
add_executable(earth_bake tools/earth_bake.c src/terrain/terrain_cube.c src/core/thread.c)
target_include_directories(earth_bake PRIVATE ${SRC_DIRS} ${stb_SOURCE_DIR})
if(MSVC)
  target_compile_options(earth_bake PRIVATE /W3 /wd4996)
else()
  target_link_libraries(earth_bake PRIVATE m)
  if(UNIX)
    target_link_libraries(earth_bake PRIVATE pthread)
  endif()
endif()
# The committed test bake (--size 256, recognisably Earth) reads back whole.
add_test(NAME earth_bake_reads COMMAND earth_bake --info ${CMAKE_CURRENT_SOURCE_DIR}/tests/data/earth_test.bake)
# `cmake --build build --target earth` stages the project and bakes it to build/earth.bake,
# with the tool's virtual environment (tools/earth/.venv) or else a python3 on the PATH;
# without either, run the two steps by hand (SOURCE.md). Not part of the default build.
set(EARTH_PY ${CMAKE_CURRENT_SOURCE_DIR}/tools/earth/.venv/Scripts/python.exe)
if(NOT EXISTS ${EARTH_PY})
  set(EARTH_PY ${CMAKE_CURRENT_SOURCE_DIR}/tools/earth/.venv/bin/python)
endif()
if(NOT EXISTS ${EARTH_PY})
  find_package(Python3 COMPONENTS Interpreter QUIET)
  set(EARTH_PY ${Python3_EXECUTABLE})
endif()
if(EARTH_PY)
  file(GLOB EARTH_TILES ${CMAKE_CURRENT_SOURCE_DIR}/data/earth/height/*.tif ${CMAKE_CURRENT_SOURCE_DIR}/data/earth/hints/*.tif)
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/earth.r16 ${CMAKE_BINARY_DIR}/earth.r16.txt
    COMMAND ${EARTH_PY} -I ${CMAKE_CURRENT_SOURCE_DIR}/tools/earth/earth.py stage ${CMAKE_CURRENT_SOURCE_DIR}/data/earth ${CMAKE_BINARY_DIR}/earth.r16
    DEPENDS ${EARTH_TILES} ${CMAKE_CURRENT_SOURCE_DIR}/data/earth/earth.json ${CMAKE_CURRENT_SOURCE_DIR}/tools/earth/earth.py
    COMMENT "Staging the Earth project" VERBATIM)
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/earth.bake
    COMMAND earth_bake ${CMAKE_BINARY_DIR}/earth.r16 ${CMAKE_BINARY_DIR}/earth.bake
    DEPENDS earth_bake ${CMAKE_BINARY_DIR}/earth.r16
    COMMENT "Baking the Earth" VERBATIM)
  add_custom_target(earth DEPENDS ${CMAKE_BINARY_DIR}/earth.bake)
endif()
